/* bubble.c — ввод массива, пузырьковая сортировка, вывод
 * Сборка: gcc -std=c11 -Wall -Wextra -O2 -o bubble bubble.c
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <limits.h>

int main(void)
{
    int n;

    printf("Введите количество элементов n: ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Ошибка: не удалось прочитать n\n");
        return 1;
    }
    if (n < 0) {
        fprintf(stderr, "Ошибка: n не может быть отрицательным\n");
        return 1;
    }
    if (n == 0) {
        printf("Массив пуст, сортировать нечего.\n");
        return 0;
    }
    /* защита от переполнения при вычислении размера блока памяти */
    if ((size_t)n > SIZE_MAX / sizeof(int)) {
        fprintf(stderr, "Ошибка: запрошен слишком большой массив\n");
        return 1;
    }

    /* динамическое выделение памяти */
    int *a = malloc((size_t)n * sizeof *a);
    if (a == NULL) {
        fprintf(stderr, "Ошибка: не удалось выделить память\n");
        return 1;
    }

    printf("Введите %d целых чисел: ", n);
    for (int i = 0; i < n; i++) {
        if (scanf("%d", &a[i]) != 1) {
            fprintf(stderr, "Ошибка: некорректный элемент №%d\n", i + 1);
            free(a);              /* освобождаем память перед выходом */
            return 1;
        }
    }

    /* --- Пузырьковая сортировка по возрастанию --- */
    for (int i = 0; i < n - 1; i++) {
        int swapped = 0;
        /* после i проходов последние i элементов уже на своих местах */
        for (int j = 0; j < n - 1 - i; j++) {
            if (a[j] > a[j + 1]) {
                int tmp  = a[j];
                a[j]     = a[j + 1];
                a[j + 1] = tmp;
                swapped  = 1;
            }
        }
        if (!swapped) {           /* обменов не было → массив отсортирован */
            break;
        }
    }

    printf("Отсортированный массив: ");
    for (int i = 0; i < n; i++) {
        printf("%d", a[i]);
        if (i + 1 < n) {
            putchar(' ');
        }
    }
    putchar('\n');

    free(a);                      /* освобождаем память */
    a = NULL;                     /* обнуляем указатель (защита от dangling pointer) */
    return 0;
}

### Как это работает

1. **Ввод `n`.** `scanf("%d", &n)` читает первое число; проверяем, что чтение удалось (`!= 1`) и что `n` не отрицательно. Случай `n == 0` обрабатывается отдельно, чтобы не вызывать `malloc(0)`.
2. **Выделение памяти.** `malloc((size_t)n * sizeof *a)` запрашивает блок под `n` элементов `int`. Приведение к `size_t` и проверка `n > SIZE_MAX / sizeof(int)` нужны, чтобы произведение размеров не переполнилось. Если `malloc` вернул `NULL` — сообщаем об ошибке и выходим, ничего не освобождая (память не была выделена).
3. **Чтение элементов.** Цикл `for` заполняет массив `a[0..n-1]`. Если пользователь ввёл не число, печатаем ошибку и **обязательно вызываем `free(a)`** перед `return` — иначе была бы утечка памяти.
4. **Пузырьковая сортировка.** Внешний цикл — это «проходы»; внутренний сравнивает соседние пары `a[j]` и `a[j+1]` и меняет их местами, если порядок нарушен. После `i`-го прохода самые большие `i` элементов уже стоят в конце, поэтому правая граница внутреннего цикла уменьшается (`n - 1 - i`). Флаг `swapped` даёт досрочный выход: если за проход не было ни одного обмена, массив уже упорядочен. Сложность — O(n²) сравнений, O(1) дополнительной памяти; сортировка устойчивая.
5. **Вывод.** Печатаем элементы через пробел; после последнего элемента пробел не выводится.
6. **Освобождение памяти.** `free(a)` возвращает блок системе, а `a = NULL` защищает от случайного использования «висячего» указателя.

Полезные замечания:
- Для больших массивов пузырьковая сортировка медленная — на практике используют `qsort` из `<stdlib.h>`, а пузырёк оставляют как учебный алгоритм.
- Проверяйте возврат `free`-обязательных путей: любая ветка выхода после `malloc` должна пройти через `free`.
