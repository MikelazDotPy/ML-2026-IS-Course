#include <stdio.h>
#include <stdlib.h>

/* Пузырьковая сортировка по возрастанию */
void bubble_sort(int *a, size_t n)
{
    for (size_t i = 0; i + 1 < n; i++) {
        int swapped = 0;                 /* оптимизация: если обменов не было, массив уже отсортирован */
        for (size_t j = 0; j + 1 < n - i; j++) {
            if (a[j] > a[j + 1]) {
                int tmp   = a[j];
                a[j]      = a[j + 1];
                a[j + 1]  = tmp;
                swapped   = 1;
            }
        }
        if (!swapped)
            break;
    }
}

int main(void)
{
    long  n_input;
    size_t n;

    /* 1. Считываем количество элементов */
    printf("Введите n: ");
    if (scanf("%ld", &n_input) != 1 || n_input <= 0) {
        fprintf(stderr, "Ошибка: n должно быть положительным целым числом.\n");
        return EXIT_FAILURE;
    }
    n = (size_t)n_input;

    /* 2. Динамическое выделение памяти */
    int *a = malloc(n * sizeof *a);
    if (a == NULL) {
        fprintf(stderr, "Ошибка: не удалось выделить память.\n");
        return EXIT_FAILURE;
    }

    /* 3. Ввод массива */
    printf("Введите %zu целых чисел: ", n);
    for (size_t i = 0; i < n; i++) {
        if (scanf("%d", &a[i]) != 1) {
            fprintf(stderr, "Ошибка: некорректный элемент на позиции %zu.\n", i + 1);
            free(a);                     /* освобождаем память перед выходом */
            return EXIT_FAILURE;
        }
    }

    /* 4. Сортировка */
    bubble_sort(a, n);

    /* 5. Вывод результата */
    printf("Отсортированный массив: ");
    for (size_t i = 0; i < n; i++)
        printf("%d%c", a[i], (i + 1 == n) ? '\n' : ' ');

    /* 6. Освобождение памяти */
    free(a);

    return EXIT_SUCCESS;
}
```

### Как это работает

1. **Ввод `n`.** `scanf` читает число в `long`, а затем проверяется, что оно положительное — иначе выход с кодом ошибки (отрицательное или нулевое значение нельзя превращать в размер массива).
2. **`malloc(n * sizeof *a)`** выделяет блок в куче ровно на `n` элементов типа `int`. Проверка на `NULL` обязательна: при большом `n` выделение может не удаться, и обращаться к памяти в этом случае нельзя.
3. **Заполнение массива.** В цикле читаются `n` чисел; `&a[i]` — это адрес i-го элемента, `a[i]` эквивалентно `*(a + i)`, так что с указателем можно работать как с обычным массивом. При ошибке ввода память освобождается до `return`.
4. **Сортировка.** Внешний проход по `i` ставит на своё итоговое место максимум из ещё не отсортированной части (он «всплывает» в конец — отсюда «пузырьковая»). Внутренний цикл до `n - i - 1` сравнивает соседние пары и меняет их местами, если порядок нарушен. Флаг `swapped` досрочно завершает работу, если за проход не было ни одного обмена — массив уже упорядочен. Сложность: O(n²) сравнений в худшем случае, O(n) в лучшем (уже отсортированный вход).
5. **Вывод.** Элементы печатаются через пробел, после последнего — перевод строки.
6. **`free(a)`** возвращает блок в кучу — ровно один вызов на один `malloc`, до завершения программы. Обнулять указатель после `free` не обязательно, так как он сразу выходит из области видимости.

### Пример работы

Введите 5 целых чисел: 4 -1 10 0 3
Отсортированный массив: -1 0 3 4 10
```

### Компиляция

```bash
gcc -std=c11 -Wall -Wextra -O2 -o sort sort.c
```

Флаги `-Wall -Wextra` включают предупреждения (полезно для отлова ошибок ввода-вывода), `-O2` — оптимизацию. Код переносимо компилируется и в C99.

Одно замечание на будущее: если `n` может быть очень большим (миллионы элементов), пузырьковая сортировка станет узким местом — тогда стоит заменить её на `qsort` из `<stdlib.h>`, который даёт O(n log n).
