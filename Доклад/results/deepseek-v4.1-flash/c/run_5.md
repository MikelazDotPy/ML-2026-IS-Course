/* bubble_sort.c — чтение массива, пузырьковая сортировка, вывод
 * Компиляция: gcc -std=c11 -Wall -Wextra -O2 -o bubble_sort bubble_sort.c
 * Запуск:     ./bubble_sort
 */

#include <stdio.h>
#include <stdlib.h>

/* Обмен значений двух элементов через указатели */
static void swap(int *a, int *b)
{
    int tmp = *a;
    *a = *b;
    *b = tmp;
}

/* Пузырьковая сортировка по возрастанию.
 * Возвращает 1, если массив уже был отсортирован (ни одного обмена). */
static int bubble_sort(int *arr, size_t n)
{
    int swapped_any = 0;

    for (size_t i = 0; i + 1 < n; i++) {
        int swapped = 0;
        /* После i проходов последние i элементов уже стоят на своих местах,
         * поэтому внутренний цикл идёт до n - 1 - i */
        for (size_t j = 0; j + 1 < n - i; j++) {
            if (arr[j] > arr[j + 1]) {
                swap(&arr[j], &arr[j + 1]);
                swapped = 1;
            }
        }
        if (swapped)
            swapped_any = 1;
        else
            break;  /* массив отсортирован — дальше проходы не нужны */
    }
    return swapped_any;
}

int main(void)
{
    int n = 0;

    printf("Введите количество элементов n: ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Ошибка: n должно быть целым числом.\n");
        return EXIT_FAILURE;
    }
    if (n <= 0) {
        fprintf(stderr, "Ошибка: n должно быть положительным (n > 0).\n");
        return EXIT_FAILURE;
    }

    /* Динамическое выделение памяти под n элементов */
    int *arr = malloc((size_t)n * sizeof *arr);
    if (arr == NULL) {
        fprintf(stderr, "Ошибка: не удалось выделить память.\n");
        return EXIT_FAILURE;
    }

    printf("Введите %d целых чисел через пробел: ", n);
    for (int i = 0; i < n; i++) {
        if (scanf("%d", &arr[i]) != 1) {
            fprintf(stderr, "Ошибка: элемент №%d введён неверно.\n", i + 1);
            free(arr);              /* освобождаем память перед выходом */
            return EXIT_FAILURE;
        }
    }

    bubble_sort(arr, (size_t)n);

    printf("Отсортированный массив: ");
    for (int i = 0; i < n; i++)
        printf("%d%c", arr[i], (i + 1 == n) ? '\n' : ' ');

    free(arr);                      /* освобождение выделенной памяти */
    return EXIT_SUCCESS;
}
```

### Как это работает

1. **Ввод `n`** — `scanf("%d", &n)` читает первое число. Проверяем, что чтение удалось и что `n > 0`; иначе сообщаем об ошибке и выходим с `EXIT_FAILURE`.
2. **Выделение памяти** — `malloc(n * sizeof(int))` резервирует память в куче под `n` целых чисел. Размер вычисляется как `(size_t)n * sizeof *arr`, что защищает от переполнения при больших `n`. Если `malloc` вернул `NULL` — выходим с ошибкой.
3. **Чтение массива** — цикл из `n` итераций читает числа в `arr[0..n-1]`. При неудачном `scanf` память освобождается через `free(arr)` и программа корректно завершается.
4. **Пузырьковая сортировка** — внешний цикл делает до `n-1` проходов. На каждом проходе внутренний цикл сравнивает соседние пары `arr[j]` и `arr[j+1]`; если левый больше правого, они меняются местами (функция `swap`). После `i`-го прохода `i` наибольших элементов уже стоят в конце, поэтому внутренний цикл идёт до `n-1-i`. Если за проход не было ни одного обмена (`swapped == 0`), массив отсортирован — цикл прерывается досрочно (оптимизация для почти отсортированных данных).
5. **Вывод** — печатаем элементы через пробел, после последнего — перевод строки (без «лишнего» пробела в конце).
6. **Освобождение памяти** — `free(arr)` возвращает память системе. Это обязательный шаг: без него при повторных запусках/в больших программах возникает утечка памяти.

**Сложность:** по времени O(n²) в худшем и среднем случае, O(n) в лучшем (при уже отсортированном массиве благодаря флагу `swapped`); по памяти — O(n) под сам массив, дополнительная память O(1).

**Пример работы:**

Введите количество элементов n: 5
Введите 5 целых чисел через пробел: 5 -3 0 12 12
Отсортированный массив: -3 0 5 12 12
```
