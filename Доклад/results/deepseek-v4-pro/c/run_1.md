#include <stdio.h>
#include <stdlib.h>

// Функция пузырьковой сортировки с оптимизацией (флаг swapped)
void bubble_sort(int arr[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int swapped = 0;
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                // Обмен элементов
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
                swapped = 1;
            }
        }
        // Если на текущем проходе не было обменов, массив уже отсортирован
        if (!swapped) break;
    }
}

int main() {
    int n;
    printf("Введите количество элементов: ");
    if (scanf("%d", &n) != 1 || n <= 0) {
        printf("Ошибка: требуется положительное целое число.\n");
        return 1;
    }

    // Динамическое выделение памяти под массив
    int *arr = (int*)malloc(n * sizeof(int));
    if (arr == NULL) {
        printf("Ошибка выделения памяти.\n");
        return 1;
    }

    printf("Введите %d целых чисел:\n", n);
    for (int i = 0; i < n; i++) {
        if (scanf("%d", &arr[i]) != 1) {
            printf("Ошибка ввода.\n");
            free(arr);
            return 1;
        }
    }

    // Сортировка пузырьком
    bubble_sort(arr, n);

    // Вывод результата
    printf("Отсортированный массив:\n");
    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    // Освобождение памяти
    free(arr);
    return 0;
}

### Как работает программа

1. **Ввод количества элементов**  
   Программа запрашивает у пользователя целое число `n` — размер массива. Если ввод некорректен или `n <= 0`, выводится сообщение об ошибке и программа завершается с кодом 1.

2. **Динамическое выделение памяти**  
   С помощью `malloc` выделяется блок памяти под `n` целых чисел. Если выделение не удалось (`malloc` вернул `NULL`), выводится сообщение об ошибке и программа завершается.

3. **Ввод элементов массива**  
   Пользователь вводит `n` целых чисел, которые сохраняются в выделенной памяти. Каждый ввод проверяется; при ошибке ввода память освобождается и программа завершается.

4. **Пузырьковая сортировка**  
   Функция `bubble_sort` последовательно сравнивает соседние элементы и меняет их местами, если они стоят в неправильном порядке.  
   После каждого прохода наибольший элемент «всплывает» в конец массива.  
   Для ускорения добавлен флаг `swapped`: если за полный проход не было ни одной перестановки, массив уже отсортирован и сортировка прекращается досрочно.

5. **Вывод результата**  
   Отсортированный массив выводится на экран в одну строку, элементы разделены пробелами.

6. **Освобождение памяти**  
   Перед завершением программы выделенная под массив память освобождается вызовом `free(arr)`, что предотвращает утечку памяти.
