#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int n, i, j, temp;
    int *arr;

    // 1. Ввод количества элементов
    printf("Введите количество чисел: ");
    if (scanf("%d", &n) != 1 || n <= 0) {
        fprintf(stderr, "Ошибка: некорректный ввод размера массива.\n");
        return 1;
    }

    // 2. Динамическое выделение памяти
    arr = (int *)malloc(n * sizeof(int));
    if (arr == NULL) {
        fprintf(stderr, "Ошибка: не удалось выделить память.\n");
        return 1;
    }

    // 3. Ввод элементов массива
    printf("Введите %d целых чисел:\n", n);
    for (i = 0; i < n; i++) {
        if (scanf("%d", &arr[i]) != 1) {
            fprintf(stderr, "Ошибка: некорректный ввод элемента.\n");
            free(arr);
            return 1;
        }
    }

    // 4. Пузырьковая сортировка (по возрастанию)
    for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - 1 - i; j++) {
            if (arr[j] > arr[j + 1]) {
                // Обмен значениями
                temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }

    // 5. Вывод отсортированного массива
    printf("Отсортированный массив:\n");
    for (i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    // 6. Освобождение памяти
    free(arr);

    return 0;
}

### Как работает программа

1. **Ввод размера** — программа запрашивает целое число `n` (количество элементов). Если `n` меньше или равно нулю или ввод некорректен, выводится сообщение об ошибке и программа завершается с ненулевым кодом.

2. **Выделение памяти** — с помощью `malloc` выделяется непрерывный блок памяти размером `n * sizeof(int)`. Адрес сохраняется в указателе `arr`. В случае неудачи (например, нехватка памяти) программа сообщает об ошибке и завершается.

3. **Ввод чисел** — пользователь вводит `n` целых чисел, которые сохраняются в ячейках массива. При ошибке ввода память освобождается и программа завершается.

4. **Пузырьковая сортировка** — стандартный алгоритм: внешний цикл выполняется `n-1` раз, внутренний проходит по несортированной части и сравнивает соседние элементы. Если текущий элемент больше следующего, они меняются местами. После каждого внешнего прохода наибольший элемент «всплывает» в конец, поэтому внутренний цикл сокращается на `i`.

5. **Вывод результата** — отсортированный массив печатается через пробел.

6. **Освобождение памяти** — после завершения работы вызван `free(arr)`, чтобы вернуть выделенную память системе. Это важно для предотвращения утечек памяти.
