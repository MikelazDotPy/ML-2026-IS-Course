#include <stdio.h>
#include <stdlib.h>

// Функция пузырьковой сортировки
void bubble_sort(int *arr, int n) {
    int swapped;
    for (int i = 0; i < n - 1; i++) {
        swapped = 0;
        for (int j = 0; j < n - 1 - i; j++) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
                swapped = 1;
            }
        }
        // Если обменов не было, массив отсортирован
        if (!swapped) break;
    }
}

int main(void) {
    int n;
    printf("Введите количество элементов: ");
    if (scanf("%d", &n) != 1 || n <= 0) {
        fprintf(stderr, "Ошибка: необходимо ввести положительное целое число.\n");
        return EXIT_FAILURE;
    }

    // Динамическое выделение памяти
    int *arr = malloc(n * sizeof(int));
    if (arr == NULL) {
        fprintf(stderr, "Ошибка: не удалось выделить память.\n");
        return EXIT_FAILURE;
    }

    printf("Введите %d целых чисел через пробел: ", n);
    for (int i = 0; i < n; i++) {
        if (scanf("%d", &arr[i]) != 1) {
            fprintf(stderr, "Ошибка ввода числа.\n");
            free(arr);
            return EXIT_FAILURE;
        }
    }

    bubble_sort(arr, n);

    printf("Отсортированный массив: ");
    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    // Освобождение памяти
    free(arr);
    return EXIT_SUCCESS;
}

## Как работает программа

1. **Ввод размера массива**  
   Пользователь вводит целое положительное число `n`. Если ввод некорректен (не число или `n ≤ 0`), программа завершается с сообщением об ошибке.

2. **Динамическое выделение памяти**  
   С помощью `malloc` выделяется блок памяти под `n` целых чисел. Если выделение не удалось (`malloc` вернул `NULL`), выводится ошибка и программа завершается.

3. **Ввод элементов**  
   Программа считывает `n` целых чисел с клавиатуры. При ошибке ввода (например, введён нечисловой символ) память освобождается и программа завершается.

4. **Пузырьковая сортировка**  
   Функция `bubble_sort` выполняет стандартную пузырьковую сортировку:
   - Внешний цикл: количество проходов `n-1`.
   - Внутренний цикл: сравнение соседних элементов до `n-1-i` (последние `i` элементов уже на своих местах).
   - Если текущий элемент больше следующего, они меняются местами.
   - Флаг `swapped` позволяет досрочно завершить сортировку, если массив уже отсортирован.

5. **Вывод результата**  
   Отсортированный массив выводится на экран через пробел.

6. **Освобождение памяти**  
   Вызов `free(arr)` возвращает выделенную память системе перед завершением программы.
